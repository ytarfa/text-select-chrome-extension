Works on code, docs and articles.

Whole-word mode skips partial hits, so renderRows stays out of the way.

import { formatDate } from './format.js';
export function renderRow(item) {
const row = document.createElement('tr');
row.append(cell(item.name), cell(formatDate(item.updatedAt)));
return row;
}
export function renderTable(items) {
const table = document.createElement('table');
for (const item of items) table.append(renderRow(item));
return table;
}
export const renderRows = (items) => items.map(renderRow);
